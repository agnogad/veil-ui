## Installation

<InstallCommand command="npx veil-ui@latest add user-profile" />

## Usage

```tsx
import { UserProfile } from "@/components/ui/user-profile";

<UserProfile
  name="Mira Chen"
  handle="@mirachen"
  bio="Product lead at Acme."
  stats={[{ label: "Followers", value: "1.2k" }]}
/>;
```
